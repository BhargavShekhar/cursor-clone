    def get_dataloaders(batch_size=64):
        transform = transforms.Compose([
            transforms.ToTensor()
        ])

        train = datasets.MNIST(
            root="data",
            train=True,
            download=True,
            transform=transform
        )

        test = datasets.MNIST(
            root="data",
            train=False,
            transform=transform
        )

        return DataLoader(train, batch_size=batch_size), \
        DataLoader(test, batch_size=batch_size)
